## Usage

```tsx
import { ProjectCard } from "@/components/project-card"
```

```tsx
<ProjectCard
  title="My Project"
  description="Project description goes here"
  status="active"
  tags={["React", "TypeScript"]}
  onViewDetails={() => console.log("View details")}
  onEdit={() => console.log("Edit")}
/>
```
